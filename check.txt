step by step process on new vm or server 

step 1 ( update server)

sudo apt update 
or 
sudo yum update 

step 2 ( install Dependency )

--> Developer → GitHub → Jenkins → Build → Artifact Repository → Docker Image → Docker Container

sudo yum install git 

(now we are installed git and now check ) 

sudo git --version 

----------------------------------------------------

now we install jenkins for that we need to install java Dependency 


sudo wget -O /etc/yum.repos.d/jenkins.repo \
    https://pkg.jenkins.io/rpm-stable/jenkins.repo
sudo dnf upgrade
# Add required dependencies for the Jenkins package
sudo dnf install fontconfig java-21-openjdk
sudo dnf install jenkins
sudo systemctl daemon-reload
sudo alternatives --set java /usr/lib/jvm/java-21-amazon-corretto/bin/java

sudo systemctl enable jenkins
sudo systemctl start jenkins
sudo systemctl status jenkins


( now you can check staus of jenkins  Active: active (running) Loaded: loaded (/usr/lib/systemd/system/jenkins.service; enabled )

---------------------------------------------------------------------
(Build Application tool maven for java)

sudo yum install maven -y

(Build Application tool for NodeJS for javascript)

sudo yum install npm 

-----------------------------------------------------------------------
->Remove Old Docker Versions

sudo yum remove docker \
                docker-client \
                docker-client-latest \
                docker-common \
                docker-latest \
                docker-latest-logrotate \
                docker-logrotate \
                docker-engine

->Set Up the Official Repository

sudo yum install -y yum-utils
sudo yum-config-manager --add-repo https://download.docker.com/linux/centos/docker-ce.repo


->Install Docker Engine

sudo yum install docker-ce docker-ce-cli containerd.io docker-buildx-plugin docker-compose-plugin
or 
sudo yum install docker

->Manage Docker as a Non-Root User 

sudo usermod -aG docker $USER
newgrp docker


->Start and Enable the Docker Service

sudo systemctl enable --now docker
sudo systemctl start docker 
sudo systemctl status docker
